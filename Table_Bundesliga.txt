function bundesliga_standings_api_key() {

}


function bundesliga_standings_config() {
    return [
        'league_id' => 175,
        'season'    => '2026/2027',
    ];
}


function bundesliga_standing_team_logo($api_logo) {
    if (!empty($api_logo)) {
        return $api_logo;
    }

    return '/wp-content/uploads/2025/09/tba.webp';
}


function bundesliga_standings_api($league_id, $season) {

    $url = 'https://apiv2.allsportsapi.com/football/?met=Standings'
        . '&APIkey=' . bundesliga_standings_api_key()
        . '&leagueId=' . $league_id
        . '&season=' . urlencode((string) $season);

    $ch = curl_init();

    curl_setopt_array($ch, [
        CURLOPT_URL            => $url,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_TIMEOUT        => 10,
        CURLOPT_CONNECTTIMEOUT => 5,
        CURLOPT_SSL_VERIFYPEER => false,
        CURLOPT_ENCODING       => 'gzip,deflate',
    ]);

    $response = curl_exec($ch);
    curl_close($ch);

    if (!$response) {
        return [];
    }

    $json  = json_decode($response, true);
    $total = $json['result']['total'] ?? [];

    return is_array($total) ? array_values($total) : [];
}


function bundesliga_standing_zone_class($team) {

    $type  = strtolower((string) ($team['standing_place_type'] ?? ''));
    $place = (int) ($team['standing_place'] ?? 0);

    if ($type !== '') {
        if (strpos($type, 'champions league') !== false) {
            return 'is-ucl';
        }
        if (strpos($type, 'europa league') !== false) {
            return 'is-uel';
        }
        if (strpos($type, 'conference') !== false) {
            return 'is-uecl';
        }
        if (strpos($type, 'relegation') !== false && strpos($type, 'play') !== false) {
            return 'is-playoff';
        }
        if (strpos($type, 'relegation') !== false) {
            return 'is-rel';
        }
    }

    if ($place >= 1 && $place <= 4) {
        return 'is-ucl';
    }
    if ($place === 5) {
        return 'is-uel';
    }
    if ($place === 6) {
        return 'is-uecl';
    }
    if ($place === 16) {
        return 'is-playoff';
    }
    if ($place >= 17) {
        return 'is-rel';
    }

    return '';
}
