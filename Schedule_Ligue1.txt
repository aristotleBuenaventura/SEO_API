function ligue1_api($leagueId, $from, $to) {

    $APIkey   = "75d4149d2873e1afd455cebbd0ec25255cea79295120d9726c22fdca3ed306be";
    $timezone = "Asia/Manila";

    $url = "https://apiv2.allsportsapi.com/football/?met=Fixtures"
        . "&APIkey={$APIkey}"
        . "&leagueId={$leagueId}"
        . "&timezone={$timezone}"
        . "&from={$from}"
        . "&to={$to}";

    $ch = curl_init();

    curl_setopt_array($ch, [
        CURLOPT_URL => $url,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_TIMEOUT => 25,
        CURLOPT_CONNECTTIMEOUT => 8,
        CURLOPT_SSL_VERIFYPEER => false,
        CURLOPT_ENCODING => "gzip,deflate"
    ]);

    $response  = curl_exec($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);

    curl_close($ch);

    if ($http_code !== 200 || !$response) {
        return [];
    }

    $json = json_decode($response, true);
    $result = $json['result'] ?? [];

    return is_array($result) ? array_values($result) : [];
}

function ligue1_team_logo($api_logo, $team_name = '') {

    }

    return '/wp-content/uploads/2025/09/tba.webp';
}


function ligue1_match_field($match, $key, $default = '') {
    if (is_array($match)) {
        return $match[$key] ?? $default;
    }
    if (is_object($match)) {
        return $match->$key ?? $default;
    }
    return $default;
}

function ligue1_schedule_shortcode() {

    $from     = "2026-08-22";
    $to       = "2027-05-31";

    $data = get_transient($cache_key);

    if ($data === false || !is_array($data) || empty($data)) {
        $data = ligue1_api($leagueId, $from, $to);

            set_transient($cache_key, $data, 30 * MINUTE_IN_SECONDS);

            delete_transient($cache_key);

        return strtotime(ligue1_match_field($a, 'event_date')) - strtotime(ligue1_match_field($b, 'event_date'));

    // upcoming matches appear when navigating next. Moves forward each day.
    // Exception: on/after the last match day — show last games at the end.
    $tz = new DateTimeZone('Asia/Manila');
    $yesterday = (new DateTime('now', $tz))->modify('-1 day')->format('Y-m-d');
    $today     = (new DateTime('now', $tz))->format('Y-m-d');

            $last_match_date = $match_date;
        }

        if ($match_date === $yesterday && $yesterday_index === null) {
            $yesterday_index = $i;
        }

        if ($match_date < $today) {
            if ($match_date !== $current_past_day) {
                $current_past_day = $match_date;
                $last_past_day_start = $i;
            }
        }
    }

        $at_last_games = true;
        $initial_slide = max(0, count($data) - 3);
    } elseif ($yesterday_index !== null) {
        $initial_slide = $yesterday_index;
    } elseif ($last_past_day_start !== null) {
        $initial_slide = $last_past_day_start;
    }

    $initial_slide = min(max(0, (int) $initial_slide), $max_start);
    $total_cards = count($data);

    <script>
    (function () {
        var root = document.getElementById(<?php echo wp_json_encode($unique_id); ?>);
        if (!root) return;

        var track = root.querySelector(".bl-track");
        var prevBtn = root.querySelector(".bl-nav-prev");
        var nextBtn = root.querySelector(".bl-nav-next");
        var cards = Array.prototype.slice.call(root.querySelectorAll(".bl-card"));
        if (!track || !cards.length) return;

        var initialIndex = parseInt(root.getAttribute("data-initial") || "0", 10) || 0;
        var atLast = root.getAttribute("data-at-last") === "1";
        var scrolling = false;

        function perView() {
            var w = window.innerWidth || document.documentElement.clientWidth;
            return w >= 768 ? 3 : 1;
        }

        function maxIndex() {
            return Math.max(0, cards.length - perView());
        }

        function currentIndex() {
            var left = track.scrollLeft;
            var best = 0;
            var bestDist = Infinity;
            for (var i = 0; i < cards.length; i++) {
                var dist = Math.abs(cards[i].offsetLeft - left);
                if (dist < bestDist) {
                    bestDist = dist;
                    best = i;
                }
            }
            return best;
        }

        function updateNav() {
            var idx = currentIndex();
            var max = maxIndex();
            prevBtn.disabled = idx <= 0;
            nextBtn.disabled = idx >= max;
        }

        function goTo(index, smooth) {
            var max = maxIndex();
            var target = Math.max(0, Math.min(index, max));
            var card = cards[target];
            if (!card) return;

            scrolling = true;
            track.scrollTo({
                left: card.offsetLeft,
                behavior: smooth === false ? "auto" : "smooth"
            });

            window.setTimeout(function () {
                scrolling = false;
                updateNav();
            }, smooth === false ? 40 : 320);
        }

        function step(dir) {
            goTo(currentIndex() + dir, true);
        }

        prevBtn.addEventListener("click", function (e) {
            e.preventDefault();
            step(-1);
        });
        nextBtn.addEventListener("click", function (e) {
            e.preventDefault();
            step(1);
        });

        var scrollTimer = null;
        track.addEventListener("scroll", function () {
            if (scrollTimer) window.clearTimeout(scrollTimer);
            scrollTimer = window.setTimeout(function () {
                if (scrolling) return;

                // Snap-lock to nearest card after finger swipe.
                var idx = currentIndex();
                var max = maxIndex();
                if (idx > max) idx = max;
                var card = cards[idx];
                if (!card) return;

                if (Math.abs(track.scrollLeft - card.offsetLeft) > 2) {
                    goTo(idx, true);
                } else {
                    updateNav();
                }
            }, 80);
        }, { passive: true });

        // Block browser back/forward edge gestures while touching the track.
        track.addEventListener("touchstart", function (e) {
            if (e.touches && e.touches.length === 1) {
                track.setAttribute("data-touching", "1");
            }
        }, { passive: true });
        track.addEventListener("touchend", function () {
            track.removeAttribute("data-touching");
            updateNav();
        }, { passive: true });

        function boot() {
            var start = initialIndex;
            if (atLast) {
                start = maxIndex();
            }
            goTo(start, false);
            updateNav();
        }

        if (document.readyState === "loading") {
            document.addEventListener("DOMContentLoaded", boot);
        } else {
            boot();
        }

        window.addEventListener("resize", function () {
            goTo(Math.min(currentIndex(), maxIndex()), false);
        });
    })();
    </script>
