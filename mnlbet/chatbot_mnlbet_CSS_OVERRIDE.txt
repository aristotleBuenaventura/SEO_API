<script>
document.addEventListener("DOMContentLoaded", function () {

    function clearWPBotStorage() {
        Object.keys(localStorage).forEach(function (key) {
            if (
                key.toLowerCase().includes("wp") ||
                key.toLowerCase().includes("chat")
            ) {
                localStorage.removeItem(key);
            }
        });

        Object.keys(sessionStorage).forEach(function (key) {
            if (
                key.toLowerCase().includes("wp") ||
                key.toLowerCase().includes("chat")
            ) {
                sessionStorage.removeItem(key);
            }
        });
    }

    function replaceLanguageWithHeaderText() {
        var header = document.querySelector(".wp-chatbot-header");
        if (!header) return;

        var langSelect = document.getElementById("wpbot_language");
        if (langSelect) {
            langSelect.remove();
        }

        if (document.querySelector(".wpbot-header-label")) return;

        var label = document.createElement("div");
        label.className = "wpbot-header-label";
        label.textContent = "MNLBET 24/7 CS";
        label.style.color = "#000";
        label.style.fontSize = "14px";
        label.style.fontWeight = "600";
        label.style.display = "flex";
        label.style.alignItems = "center";
        label.style.marginLeft = "10px";

        header.prepend(label);
    }

    function hardResetChat() {
        var tries = 0;

        var interval = setInterval(function () {

            var msgContainer = document.getElementById("wp-chatbot-messages-container");
            var resetBtn = document.getElementById("wp-chatbot-desktop-reload");

            if (msgContainer) {
                msgContainer.innerHTML = "";
            }

            if (resetBtn) {
                resetBtn.click();
                clearInterval(interval);
            }

            tries++;
            if (tries > 50) clearInterval(interval);

        }, 100);
    }

    /* Siguraduhing CLOSED muna on load */
    function ensureChatClosed() {
        var chatBall = document.getElementById("wp-chatbot-ball");
        var chatBoard = document.getElementById("wp-chatbot-board-container");

        if (chatBall && chatBoard && chatBoard.classList.contains("active-chat-board")) {
            chatBall.click();
        }
    }

    function autoOpen() {
        setTimeout(function () {

            var chatBall = document.getElementById("wp-chatbot-ball");
            var chatBoard = document.getElementById("wp-chatbot-board-container");

            if (!chatBall || !chatBoard) return;

            if (!chatBoard.classList.contains("active-chat-board")) {
                chatBall.click();
            }

        }, 5000);
    }

    clearWPBotStorage();
    replaceLanguageWithHeaderText();

    /* Close muna kung accidentally open */
    ensureChatClosed();
    setTimeout(ensureChatClosed, 500);
    setTimeout(ensureChatClosed, 1500);

    hardResetChat();
    autoOpen();

});
</script>
